# Outward-action patterns for the Laurelin approval gate (plan section 6).
# One rule per line:  <name>: <regular expression>   (case-insensitive)
# A tool call whose command text matches any rule waits for Joe.
# Lines starting with # are comments. Blank lines are ignored.

# Deploys, pushes, merges, releases
deploy.vercel: \bvercel\b(?!\s+(--help|-h|ls|list|whoami|inspect|logs|env\s+(ls|pull)))
deploy.wisp: \bwisp\s+(deploy|publish)\b
deploy.gh-merge: \bgh\s+pr\s+merge\b
deploy.gh-release: \bgh\s+release\s+(create|upload|edit|delete)\b
deploy.git-push: \bgit\s+push\b
deploy.npm-publish: \bnpm\s+publish\b
deploy.gcloud: \bgcloud\s+.*\b(deploy|run\s+deploy|functions\s+deploy)\b
deploy.fly: \b(fly|flyctl)\s+deploy\b

# Email sends
email.cli: \b(sendmail|mailx?\s+-s|mutt\s+-s|swaks)\b
email.tool: \b(send_?email|send_?message|send_?draft|gmail\.users\.messages\.send|messages/send|reply_?all|forward_?email)\b
email.smtp: \bsmtp(s)?://
email.superhuman: \bsuperhuman\b.*\bsend\b
email.superhuman-actions: \b(send_draft|undo_send|create_or_update_event|respond_to_event|unsubscribe|mark_spam|trash_thread|forward)\b

# Posts and messages to social or chat services
post.social: \b(api\.(x|twitter)\.com|x\.com/i/api|tweet|post_?tweet|substack\.com/api|linkedin\.com/v2/(ugcPosts|posts|shares)|graph\.instagram\.com|instagram\.com/api)\b
post.chat: \b(discord(app)?\.com/api/webhooks|discord\.com/api/v\d+/channels|osascript.*Messages|imessage|send_?imessage|slack\.com/api/chat\.postMessage)\b

# Money
pay.stripe-live: \bsk_live_
pay.stripe-cli: \bstripe\s+(charges|payment_?intents|invoices|payouts|transfers)\s+create\b
pay.generic: \b(create_?charge|create_?payment|purchase|checkout\.session|transfer_?funds|place_?order|buy_?now)\b
pay.boat: \bboat\s+(prompt|create|start|up)\b
pay.cloud-gpu: \b(runpod|lambda-?labs|vast\.ai|modal\s+deploy|replicate\.com/v1/predictions)\b

# DNS and Workspace changes
dns.records: \b(route53\s+change-resource-record-sets|cloudflare.*(dns_records|zones/.*/dns)|nsupdate|vercel\s+dns\s+(add|rm)|dnscontrol\s+push|namecheap.*setHosts)\b
dns.workspace: \b(admin\.googleapis\.com|gam\s+(create|update|delete)\s)\b

# Form submissions (applications, portals, journals)
form.submit: \b(curl|wget|http|xh)\b[^\n]*\b(-X\s*POST|--data|--form|-F\s|-d\s)[^\n]*\b(submit|apply|application|portal|arxiv\.org/submit|manuscript|register)\b
form.tool: \b(submit_?form|form_?submit|click_?submit|submit_?application)\b
